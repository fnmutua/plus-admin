 SELECT 
    c.id AS county_id,
    c.name AS county_name,
    sc.id AS subcounty_id,
    sc.name AS subcounty_name,
    w.id AS ward_id,
    w.name AS ward_name
FROM 
    county c
JOIN 
    subcounty sc ON sc.county_id = c.id
JOIN 
    ward  w ON w.subcounty_id = sc.id
ORDER BY 
    c.name, sc.name, w.name;

SET filters = (
  SELECT array_agg(
    CASE
      WHEN elem->>'field' = 'settlement_type' THEN
        jsonb_set(elem, '{value}', '["Slum"]'::jsonb)
      ELSE
        elem
    END
  )::jsonb[]
  FROM unnest(filters) AS elem
)
WHERE EXISTS (
  SELECT 1
  FROM unnest(filters) AS elem
  WHERE elem->>'field' = 'settlement_type' AND elem->>'value' = '[1]'
